        function handleLogin(event) {
            event.preventDefault();
            document.getElementById('login-screen').classList.add('hidden');
            document.getElementById('dashboard-screen').classList.remove('hidden');
            initChart();
            renderDashboard();
        }

        function handleLogout() {
            document.getElementById('dashboard-screen').classList.add('hidden');
            document.getElementById('login-screen').classList.remove('hidden');
        }

        function toggleSystemLock() {
            isSystemLocked = !isSystemLocked;
            const btn = document.getElementById('lock-btn');
            const txt = document.getElementById('lock-text');
            const badge = document.getElementById('lock-badge');
            
            if (isSystemLocked) {
                btn.className = "bg-rose-500/20 text-rose-300 px-3 py-2 rounded-xl text-xs font-semibold transition border border-rose-500/30 flex items-center space-x-1.5";
                txt.innerText = "Locked";
                badge.classList.remove('hidden');
                addNotification("System Security", "Master lock engaged. Remote alterations disabled.");
            } else {
                btn.className = "bg-slate-800 hover:bg-slate-700 text-slate-300 px-3 py-2 rounded-xl text-xs font-semibold transition border border-slate-700 flex items-center space-x-1.5";
                txt.innerText = "Unlocked";
                badge.classList.add('hidden');
                addNotification("System Security", "Master lock disengaged.");
            }
            renderDashboard();
        }

        function switchCarrier(carrier) {
            currentCarrier = carrier;
            const badge = document.getElementById('signal-health-badge');
            if(carrier.includes('Jio')) badge.innerText = "RSSI: -78dBm (Strong)";
            else if(carrier.includes('Airtel')) badge.innerText = "RSSI: -84dBm (Good)";
            else badge.innerText = "RSSI: -92dBm (Fair)";
            
            addNotification("Carrier Switched", `Migrated eSIM profile to ${carrier}`);
            renderDashboard();
        }

        function toggleNotifications() {
            document.getElementById('notif-panel').classList.toggle('translate-x-full');
        }

        function clearNotifications() {
            document.getElementById('notif-list').innerHTML = `<p class="text-slate-500 text-center py-4">No new alerts</p>`;
            document.getElementById('notif-count').innerText = "0";
        }

        function addNotification(title, desc) {
            const list = document.getElementById('notif-list');
            const item = document.createElement('div');
            item.className = "bg-slate-950 p-3 rounded-xl border border-slate-800";
            item.innerHTML = `<p class="font-semibold text-slate-200">${title}</p><p class="text-[10px] text-slate-500 mt-0.5">${desc}</p><span class="text-[9px] text-emerald-400 mt-1 block">Just now</span>`;
            list.prepend(item);
            const count = document.getElementById('notif-count');
            count.innerText = parseInt(count.innerText) + 1;
        }

        function triggerScene(sceneKey) {
            if (isSystemLocked) return;
            if (sceneKey === 'leaving') {
                appliances.forEach(a => a.isOn = false);
                addNotification("Scene Executed", "Leaving Home: All appliances powered OFF.");
            } else if (sceneKey === 'night') {
                appliances.forEach(a => {
                    if (a.type === 'light') { a.isOn = true; a.brightness = 15; a.rgb = "#3b82f6"; }
                    else if (a.type === 'air-conditioner') { a.isOn = true; a.temp = 26; a.mode = "Eco"; }
                    else { a.isOn = false; }
                });
                addNotification("Scene Executed", "Good Night: Bedroom dimming & Eco AC active.");
            } else if (sceneKey === 'saver') {
                appliances.forEach(a => {
                    if (a.type === 'heater') a.isOn = false;
                    if (a.type === 'air-conditioner') a.temp = 25;
                });
                addNotification("Scene Executed", "Energy Saver: Heavy loads optimized.");
            }
            renderDashboard();
        }

            let totalWattageCalc = 0;

            const filtered = appliances.filter(app => {
                const matchesSearch = app.name.toLowerCase().includes(searchQuery);
                const matchesCategory = filterCategory === 'all' || app.type === filterCategory;
                return matchesSearch && matchesCategory;
            });

            filtered.forEach(app => {
                if (app.isOn) {
                    activeCount++;
                    totalWattageCalc += app.wattage;
                }

                const card = document.createElement('div');
                card.className = `bg-slate-900 border ${app.isOn ? 'border-emerald-500/50 shadow-lg shadow-emerald-950/40' : 'border-slate-800'} rounded-2xl p-5 flex flex-col justify-between transition-all duration-300`;
                
                let inlineControls = '';
                if (app.type === 'air-conditioner') {
                    inlineControls = `
                        <div class="mt-4 pt-3 border-t border-slate-800 space-y-2 text-xs">
                            <div class="flex justify-between text-slate-400"><span>Temperature:</span><span class="font-bold text-emerald-400">${app.temp}°C</span></div>
                            <input type="range" min="16" max="30" value="${app.temp}" ${(!app.isOn || isSystemLocked) ? 'disabled' : ''} onchange="updateProp(${app.id}, 'temp', this.value)" class="w-full accent-emerald-500">
                            <div class="flex space-x-1">
                                ${['Cool', 'Dry', 'Eco'].map(m => `<button onclick="updateProp(${app.id}, 'mode', '${m}')" class="flex-1 py-1 rounded text-[10px] font-semibold border ${app.mode === m ? 'bg-emerald-600 text-white border-emerald-500' : 'bg-slate-950 text-slate-400 border-slate-800'}">${m}</button>`).join('')}
                            </div>
                        </div>
                    `;
                } else if (app.type === 'fan') {
                    inlineControls = `
                        <div class="mt-4 pt-3 border-t border-slate-800 text-xs">
                            <div class="flex justify-between text-slate-400 mb-1.5"><span>Speed Level:</span><span class="font-bold text-emerald-400">Level ${app.speed}</span></div>
                            <div class="flex space-x-2">
                                ${[1, 2, 3, 4].map(s => `<button onclick="updateProp(${app.id}, 'speed',${s})" class="flex-1 py-1 rounded text-xs font-semibold border ${app.speed === s ? 'bg-emerald-600 text-white border-emerald-500' : 'bg-slate-950 text-slate-400 border-slate-800'}">${s}</button>`).join('')}
                            </div>
                        </div>
                    `;
                } else if (app.type === 'light') {
                    inlineControls = `
                        <div class="mt-4 pt-3 border-t border-slate-800 space-y-2 text-xs">
                            <div class="flex justify-between text-slate-400"><span>Brightness:</span><span class="font-bold text-emerald-400">${app.brightness}%</span></div>
                            <input type="range" min="10" max="100" value="${app.brightness}" ${(!app.isOn || isSystemLocked) ? 'disabled' : ''} onchange="updateProp(${app.id}, 'brightness', this.value)" class="w-full accent-emerald-500">
                            <div class="flex items-center justify-between">
                                <span class="text-slate-400">RGB Color:</span>
                                <input type="color" value="${app.rgb}" ${(!app.isOn || isSystemLocked) ? 'disabled' : ''} onchange="updateProp(${app.id}, 'rgb', this.value)" class="bg-transparent cursor-pointer h-6 w-10">
                            </div>
                        </div>
                    `;
                }

                card.innerHTML = `
                    <div>
                        <div class="flex justify-between items-start mb-3">
                            <div class="p-3 rounded-xl ${app.isOn ? 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/20' : 'bg-slate-800 text-slate-500'}">
                                <i class="fa-solid ${app.icon} text-lg"></i>
                            </div>
                            <span class="px-2.5 py-0.5 text-[10px] font-bold rounded-full ${app.isOn ? 'bg-emerald-500/20 text-emerald-300 border border-emerald-500/30' : 'bg-slate-800 text-slate-400'}">
                                ${app.isOn ? 'ONLINE / ON' : 'OFFLINE / OFF'}
                            </span>
                        </div>

                        <h3 class="text-sm font-bold text-slate-100">${app.name}</h3>
                        <div class="text-[11px] text-slate-400 mt-1 space-y-0.5">
                            <p>eSIM ICCID: <span class="font-mono text-slate-300">${app.iccid}</span></p>
                            <p>Data Used: <span class="text-blue-400 font-mono">${app.dataUsage}</span></p>
                        </div>

                        ${inlineControls}

                        <div class="mt-4 pt-3 border-t border-slate-800/80 flex justify-between text-[11px] text-slate-400">
                            <span><i class="fa-solid fa-tower-broadcast text-blue-400 mr-1"></i>${app.signal}</span>
                            <span class="bg-slate-950 px-2 py-0.5 rounded text-slate-300 border border-slate-800">${app.protocol}</span>
                        </div>
                    </div>

                    <div class="mt-5">
                        <button onclick="toggleAppliance(${app.id})" ${isSystemLocked ? 'disabled' : ''} class="w-full py-2 px-4 rounded-xl font-semibold text-xs transition flex items-center justify-center space-x-2 ${isSystemLocked ? 'opacity-50 cursor-not-allowed bg-slate-800 text-slate-500' : (app.isOn ? 'bg-rose-500/10 text-rose-300 hover:bg-rose-500/20 border border-rose-500/30' : 'bg-emerald-600 text-white hover:bg-emerald-500 shadow-md shadow-emerald-600/20')}">
                            <i class="fa-solid fa-power-off"></i>
                            <span>${app.isOn ? 'Turn OFF Remotely' : 'Turn ON Remotely'}</span>
                        </button>
                    </div>
                `;
                grid.appendChild(card);
            });

            document.getElementById('stat-active-count').innerText = `${activeCount} / ${appliances.length}`;
            document.getElementById('total-wattage').innerText = `${totalWattageCalc} W`;
            
            const estMonthlyCost = Math.round((totalWattageCalc * 24 * 30 / 1000) * 7);
            document.getElementById('est-monthly-cost').innerText = `₹${estMonthlyCost.toLocaleString()}`;

            const loadText = document.getElementById('load-status-text');
            if (totalWattageCalc > 3000) {
                loadText.innerText = "High Load Alert!";
                loadText.className = "text-xs text-rose-400 font-bold animate-pulse";
            } else {
                loadText.innerText = "Normal Load";
                loadText.className = "text-xs text-slate-500";
            }
        }

        function toggleAppliance(id) {
            if (isSystemLocked) return;
            const target = appliances.find(a => a.id === id);
            if (target) {
                target.isOn = !target.isOn;
                renderDashboard();
            }
        }

        function toggleAll(state) {
            if (isSystemLocked) return;
            appliances.forEach(a => a.isOn = state);
            renderDashboard();
        }

        function updateProp(id, prop, val) {
            const target = appliances.find(a => a.id === id);
            if (target) {
                target[prop] = val;
                renderDashboard();
            }
        }

        function openModal() {
            document.getElementById('addModal').classList.remove('hidden');
            document.getElementById('addModal').classList.add('flex');
        }

        function closeModal() {
            document.getElementById('addModal').classList.remove('flex');
            document.getElementById('addModal').classList.add('hidden');
        }

        function handleFormSubmit(event) {
            event.preventDefault();
            const name = document.getElementById('appName').value;
            const type = document.getElementById('appType').value;
            const protocol = document.getElementById('appProtocol').value;

            let icon = 'fa-plug', wattage = 100;
            if(type === 'air-conditioner') { icon = 'fa-snowflake'; wattage = 1500; }
            else if(type === 'fan') { icon = 'fa-fan'; wattage = 70; }
            else if(type === 'light') { icon = 'fa-lightbulb'; wattage = 20; }
            else if(type === 'heater') { icon = 'fa-fire-flame-curved'; wattage = 2000; }

            appliances.push({
                id: appliances.length + 1,
                name: name, type: type, icon: icon, isOn: false,
                iccid: `89914...${Math.floor(10000 + Math.random() * 90000)}`,
                signal: "-80 dBm", protocol: protocol, dataUsage: "0.5 MB",
                wattage: wattage, temp: 24, mode: "Cool", speed: 2, brightness: 100, rgb: "#10b981"
            });

            closeModal();
            renderDashboard();
        }
